     1→/** FIELD_REGISTRY · 当前 AI/结构化采纳允许写入字段的单一事实源。 */
     2→import type { FieldSpec } from './types'
     3→
     4→const trimString = (val: unknown): unknown =>
     5→  typeof val === 'string' ? val.trim() : val
     6→
     7→function text(target: string, field: string, labels?: string[]): FieldSpec {
     8→  return { target, field, type: 'string', labels, sanitize: trimString }
     9→}
    10→
    11→function longtext(target: string, field: string, labels?: string[]): FieldSpec {
    12→  return { target, field, type: 'longtext', labels, sanitize: trimString }
    13→}
    14→
    15→function num(target: string, field: string, labels?: string[]): FieldSpec {
    16→  return { target, field, type: 'number', labels }
    17→}
    18→
    19→function bool(target: string, field: string, labels?: string[]): FieldSpec {
    20→  return { target, field, type: 'boolean', labels }
    21→}
    22→
    23→function json(target: string, field: string, labels?: string[]): FieldSpec {
    24→  return { target, field, type: 'json', labels }
    25→}
    26→
    27→/** IndexedDB 原生对象字段（区别于以 JSON string 存储的 json 字段）。 */
    28→function object(target: string, field: string, labels?: string[]): FieldSpec {
    29→  return { target, field, type: 'object', labels }
    30→}
    31→
    32→function arr(target: string, field: string, labels?: string[]): FieldSpec {
    33→  return { target, field, type: 'array', labels }
    34→}
    35→
    36→function enumeration(
    37→  target: string,
    38→  field: string,
    39→  enums: string[],
    40→  labels?: string[],
    41→): FieldSpec {
    42→  return { target, field, type: 'enum', enums, labels, sanitize: trimString }
    43→}
