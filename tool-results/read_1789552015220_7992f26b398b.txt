    78→function readIdArray(value: unknown, path: string): number[] | undefined {
    79→  if (value === undefined) return undefined
    80→  const result = readArray(value, path).map((item, index) => readInteger(item, `${path}[${index}]`, { min: 1 }))
    81→  if (new Set(result).size !== result.length) failSchema('duplicate_value', path, '不得包含重复 ID')
    82→  return result
    83→}

    85→function readWriteTarget(value: unknown, path: string): AgentRunWriteTargetV1 {
    86→  const record = readRecord(value, path)
    87→  assertExactKeys(record, ['table', 'fields', 'mode', 'adoptionExtension'], ['table', 'fields', 'mode'], path)
    88→  const table = readString(record.table, `${path}.table`, { max: 100 })
    89→  if (!REGISTRY_BY_NAME.has(table)) failSchema('unknown_table', `${path}.table`, `未登记的数据表 ${table}`)
    90→  const fields = readArray(record.fields, `${path}.fields`).map((item, index) => (
    91→    readString(item, `${path}.fields[${index}]`, { max: 100 })
    92→  ))
    93→  assertUnique(fields, `${path}.fields`)
    94→  const mode = readEnum(record.mode, WRITE_MODES, `${path}.mode`)
    95→  const adoptionExtension = record.adoptionExtension === undefined
    96→    ? undefined
    97→    : readString(record.adoptionExtension, `${path}.adoptionExtension`, { max: 120 })
    98→  const extension = adoptionExtension
    99→    ? ADOPTION_EXTENSIONS.find(item => item.id === adoptionExtension)
   100→    : undefined
   101→  if (adoptionExtension && (!extension || extension.target !== table)) {
   102→    failSchema(
   103→      'unknown_adoption_extension',
   104→      `${path}.adoptionExtension`,
   105→      `未登记或与目标表不匹配的采纳扩展 ${adoptionExtension}`,
   106→    )
   107→  }
   108→  if (mode === 'none' && fields.length > 0) {
   109→    failSchema('invalid_permission', path, 'mode=none 时不得声明可写字段')
   110→  }
   111→  if (mode !== 'none' && fields.length === 0 && !adoptionExtension) {
   112→    failSchema('invalid_permission', path, '可写目标必须声明至少一个字段')
   113→  }
   114→  const registeredFields = new Set((FIELD_BY_TARGET.get(table) ?? []).map(field => field.field))
   115→  for (const field of fields) {
   116→    if (!registeredFields.has(field)) {
   117→      failSchema('unknown_write_field', `${path}.fields`, `${table}.${field} 未在 FIELD_REGISTRY 登记`)
   118→    }
   119→  }
   120→  return {
   121→    table,
   122→    fields,
   123→    mode,
   124→    ...(adoptionExtension ? { adoptionExtension } : {}),
   125→  }
   126→}

   154→function readExecutionBinding(value: unknown, path: string): AgentRunStepExecutionBindingV1 {
   155→  const record = readRecord(value, path)
   156→  const keys = [
   157→    'stepId',
   158→    'version',
   159→    'skillId',
   160→    'skillVersion',
   161→    'promptVersion',
   162→    'toolSchemaVersion',
   163→    'toolSchemaHash',
   164→    'promptExecution',
   165→    'formalEntry',
   166→  ] as const
   167→  assertExactKeys(record, keys, keys.filter(key => key !== 'promptExecution' && key !== 'formalEntry'), path)
   168→  if (record.version !== 1 || record.skillVersion !== 1) {
   169→    failSchema('unsupported_version', path, '仅支持 execution binding v1 / Skill v1')
   170→  }
   171→  let promptExecution: AgentRunStepExecutionBindingV1['promptExecution']
   172→  if (record.promptExecution !== undefined) {
   173→    const prompt = readRecord(record.promptExecution, `${path}.promptExecution`)
   174→    const promptKeys = [
   175→      'version', 'moduleKey', 'templateId', 'templateName', 'templateScope',
   176→      'templateUpdatedAt', 'templateHash', 'parameterValuesHash', 'overridesHash',
   177→    ] as const
   178→    assertExactKeys(prompt, promptKeys, promptKeys, `${path}.promptExecution`)
   179→    if (prompt.version !== 1) failSchema('unsupported_version', `${path}.promptExecution.version`, '仅支持 v1')
   180→    if (prompt.templateScope !== 'system' && prompt.templateScope !== 'user') {
   181→      failSchema('invalid_value', `${path}.promptExecution.templateScope`, '必须是 system 或 user')
   182→    }
   183→    promptExecution = {
   184→      version: 1,
   185→      moduleKey: readString(prompt.moduleKey, `${path}.promptExecution.moduleKey`, { max: 160 }),
   186→      templateId: prompt.templateId === null
   187→        ? null
   188→        : readInteger(prompt.templateId, `${path}.promptExecution.templateId`, { min: 1 }),
   189→      templateName: readString(prompt.templateName, `${path}.promptExecution.templateName`, { max: 500 }),
   190→      templateScope: prompt.templateScope,
   191→      templateUpdatedAt: readInteger(prompt.templateUpdatedAt, `${path}.promptExecution.templateUpdatedAt`),
   192→      templateHash: readHash(prompt.templateHash, `${path}.promptExecution.templateHash`),
   193→      parameterValuesHash: readHash(prompt.parameterValuesHash, `${path}.promptExecution.parameterValuesHash`),
   194→      overridesHash: readHash(prompt.overridesHash, `${path}.promptExecution.overridesHash`),
   195→    }
   196→  }
   197→  let formalEntry: AgentRunStepExecutionBindingV1['formalEntry']
   198→  if (record.formalEntry !== undefined) {
   199→    const entry = readRecord(record.formalEntry, `${path}.formalEntry`)
   200→    const entryKeys = ['version', 'entryId', 'bindingJson', 'bindingHash'] as const
   201→    assertExactKeys(entry, entryKeys, entryKeys, `${path}.formalEntry`)
   202→    if (entry.version !== 1) failSchema('unsupported_version', `${path}.formalEntry.version`, '仅支持 v1')
   203→    formalEntry = {
   204→      version: 1,
   205→      entryId: readString(entry.entryId, `${path}.formalEntry.entryId`, { max: 160 }),
   206→      bindingJson: readString(entry.bindingJson, `${path}.formalEntry.bindingJson`, { max: 100_000 }),
   207→      bindingHash: readHash(entry.bindingHash, `${path}.formalEntry.bindingHash`),
   208→    }
   209→  }
   210→  return {
   211→    stepId: readString(record.stepId, `${path}.stepId`, { max: 160 }),
   212→    version: 1,
   213→    skillId: readString(record.skillId, `${path}.skillId`, { max: 160 }),
   214→    skillVersion: 1,
   215→    promptVersion: readString(record.promptVersion, `${path}.promptVersion`, { max: 160 }),
   216→    toolSchemaVersion: readString(record.toolSchemaVersion, `${path}.toolSchemaVersion`, { max: 160 }),
   217→    toolSchemaHash: readHash(record.toolSchemaHash, `${path}.toolSchemaHash`),
   218→    ...(promptExecution ? { promptExecution } : {}),
   219→    ...(formalEntry ? { formalEntry } : {}),
   220→  }
   221→}

   223→function readExecutionBindingV2(value: unknown, path: string): AgentRunStepExecutionBindingV2 {
   224→  const record = readRecord(value, path)
   225→  const keys = [
   226→    'stepId',
   227→    'version',
   228→    'skillId',
   229→    'skillVersion',
   230→    'skillDefinitionJson',
   231→    'skillDefinitionHash',
   232→    'contextAccessPolicyHash',
   233→    'promptVersion',
   234→    'toolSchemaVersion',
   235→    'toolSchemaHash',
   236→    'contextSourceKeys',
   237→    'optionalContextActivations',
   238→    'writeTargets',
   239→    'maxOutputTokens',
   240→    'formalEntry',
   241→  ] as const
   242→  assertExactKeys(record, keys, keys.filter(key => key !== 'formalEntry'), path)
   243→  if (record.version !== 2 || record.skillVersion !== 2) {
   244→    failSchema('unsupported_version', path, '仅支持 execution binding v2 / Skill snapshot v2')
   245→  }
   246→  const contextSourceKeys = readArray(record.contextSourceKeys, `${path}.contextSourceKeys`)
   247→    .map((item, index) => readString(item, `${path}.contextSourceKeys[${index}]`, { max: 160 }))
   248→  assertUnique(contextSourceKeys, `${path}.contextSourceKeys`)
   249→  for (const sourceKey of contextSourceKeys) {
   250→    if (!CONTEXT_SOURCE_BY_KEY.has(sourceKey)) {
   251→      failSchema('unknown_context_source', `${path}.contextSourceKeys`, `未登记的上下文源 ${sourceKey}`)
   252→    }
   253→  }
   254→  const optionalContextActivations = readArray(
   255→    record.optionalContextActivations,
   256→    `${path}.optionalContextActivations`,
   257→  ).map((item, index) => {
   258→    const activationPath = `${path}.optionalContextActivations[${index}]`
   259→    const activation = readRecord(item, activationPath)
   260→    assertExactKeys(
   261→      activation,
   262→      ['sourceKey', 'reasonCode', 'boundaryHash'],
   263→      ['sourceKey', 'reasonCode'],
   264→      activationPath,
   265→    )
   266→    return {
   267→      sourceKey: readString(activation.sourceKey, `${activationPath}.sourceKey`, { max: 160 }),
   268→      reasonCode: readEnum(
   269→        activation.reasonCode,
   270→        ['perspective-character', 'prior-outline-candidate', 'explicit-runtime-boundary'],
   271→        `${activationPath}.reasonCode`,
   272→      ),
   273→      ...(activation.boundaryHash === undefined ? {} : {
   274→        boundaryHash: readHash(activation.boundaryHash, `${activationPath}.boundaryHash`),
   275→      }),
   276→    }
   277→  })
   278→  assertUnique(optionalContextActivations.map(item => item.sourceKey), `${path}.optionalContextActivations`)
   279→  for (const activation of optionalContextActivations) {
   280→    if (!contextSourceKeys.includes(activation.sourceKey)) {
   281→      failSchema('invalid_execution_binding', path, `激活来源 ${activation.sourceKey} 未进入实际上下文集合`)
   282→    }
   283→  }
   284→  const writeTargets = readArray(record.writeTargets, `${path}.writeTargets`)
   285→    .map((item, index) => readWriteTarget(item, `${path}.writeTargets[${index}]`))
   286→  assertUnique(
   287→    writeTargets.map(item => `${item.table}:${item.adoptionExtension ?? ''}`),
   288→    `${path}.writeTargets`,
   289→  )
   290→  const formalEntry = record.formalEntry === undefined
   291→    ? undefined
   292→    : readExecutionBinding({
   293→      stepId: record.stepId,
   294→      version: 1,
   295→      skillId: record.skillId,
   296→      skillVersion: 1,
   297→      promptVersion: record.promptVersion,
   298→      toolSchemaVersion: record.toolSchemaVersion,
   299→      toolSchemaHash: record.toolSchemaHash,
   300→      formalEntry: record.formalEntry,
   301→    }, `${path}.formalEntryCarrier`).formalEntry
   302→  return {
   303→    stepId: readString(record.stepId, `${path}.stepId`, { max: 160 }),
   304→    version: 2,
   305→    skillId: readString(record.skillId, `${path}.skillId`, { max: 160 }),
   306→    skillVersion: 2,
   307→    skillDefinitionJson: readString(record.skillDefinitionJson, `${path}.skillDefinitionJson`, { max: 100_000 }),
   308→    skillDefinitionHash: readHash(record.skillDefinitionHash, `${path}.skillDefinitionHash`),
   309→    contextAccessPolicyHash: readHash(record.contextAccessPolicyHash, `${path}.contextAccessPolicyHash`),
   310→    promptVersion: readString(record.promptVersion, `${path}.promptVersion`, { max: 160 }),
   311→    toolSchemaVersion: readString(record.toolSchemaVersion, `${path}.toolSchemaVersion`, { max: 160 }),
   312→    toolSchemaHash: readHash(record.toolSchemaHash, `${path}.toolSchemaHash`),
   313→    contextSourceKeys,
   314→    optionalContextActivations,
   315→    writeTargets,
   316→    maxOutputTokens: readInteger(record.maxOutputTokens, `${path}.maxOutputTokens`, { min: 1 }),
   317→    ...(formalEntry ? { formalEntry } : {}),
   318→  }
   319→}

   339→function parseAgentRunContractV1Base(

   348→    'contract',
   349→  )

   357→      parentRunId: readInteger(ownershipRecord.parentRunId, 'contract.ownership.parentRunId', { min: 1 }),

   359→    }
   360→  }
   361→
   362→  let lineage: AgentRunContractV1['lineage']
   363→  if (record.lineage !== undefined) {
   364→    const lineageRecord = readRecord(record.lineage, 'contract.lineage')

   382→  let runtime: AgentRunContractV1['scope']['runtime']

   387→      ['productRuntimeSessionId', 'baseSequence', 'stateHash', 'visibilityHash', 'releaseHash'],

   393→      baseSequence: readInteger(runtimeRecord.baseSequence, 'contract.scope.runtime.baseSequence', { min: 0 }),

   419→
   420→  const permissionsRecord = readRecord(record.permissions, 'contract.permissions')
   421→  assertExactKeys(
   422→    permissionsRecord,
   423→    ['contextSourceKeys', 'writeTargets'],

   438→    }
   439→  }
   440→  const writeTargets = readArray(
   441→    permissionsRecord.writeTargets,
   442→    'contract.permissions.writeTargets',
   443→  ).map((item, index) => readWriteTarget(item, `contract.permissions.writeTargets[${index}]`))
   444→  assertUnique(writeTargets.map(target => target.table), 'contract.permissions.writeTargets')
   445→  const executionBindings = record.executionBindings === undefined
   446→    ? undefined
   447→    : readArray(record.executionBindings, 'contract.executionBindings')
   448→        .map((item, index) => readExecutionBinding(item, `contract.executionBindings[${index}]`))
   449→  if (executionBindings?.length === 0) {
   450→    failSchema('invalid_execution_binding', 'contract.executionBindings', '存在字段时不得为空')
   451→  }
   452→  if (executionBindings) assertUnique(executionBindings.map(item => item.stepId), 'contract.executionBindings')
   453→  if (!bindingCarriedBySkillSnapshot && record.runtimeBindingHash === undefined && executionBindings === undefined) {
   454→    failSchema(
   455→      'missing_execution_binding',

   457→      '当前 RunContract 必须冻结 runtimeBindingHash 或 executionBindings',
   458→    )
   459→  }
   460→
   461→  let dependencyReceiptPolicy: AgentRunContractV1['dependencyReceiptPolicy']
   462→  if (record.dependencyReceiptPolicy !== undefined) {
   463→    const policy = readRecord(record.dependencyReceiptPolicy, 'contract.dependencyReceiptPolicy')
   464→    assertExactKeys(
   465→      policy,
   466→      ['requiredForJoin', 'verifierSetVersion'],
   467→      ['requiredForJoin', 'verifierSetVersion'],
   468→      'contract.dependencyReceiptPolicy',
   469→    )
   470→    if (policy.requiredForJoin !== true) {
   471→      failSchema(

   725→function writePermissionSignatures(targets: readonly AgentRunWriteTargetV1[]): string[] {
   726→  const signatures: string[] = []
   727→  for (const target of targets) {
   728→    if (target.fields.length === 0) {
   729→      signatures.push(`${target.table}:${target.adoptionExtension ?? ''}:${target.mode}:#extension`)
   730→      continue
   731→    }
   732→    for (const field of target.fields) {
   733→      signatures.push(`${target.table}:${target.adoptionExtension ?? ''}:${target.mode}:${field}`)
   734→    }
   735→  }
   736→  return [...new Set(signatures)].sort()
   737→}

   739→function parseSkillBoundContract(
   740→  value: unknown,
   741→  expectedVersion: 2 | 3,
   742→): {
   743→  base: AgentRunContractV1
   744→  executionBindings: AgentRunStepExecutionBindingV2[]
   745→  record: Record<string, unknown>
   746→} {
   747→  const record = readRecord(value, 'contract')
   748→  const allowedKeys = [
   749→    'version',
   750→    'objective',
   751→    'workflowKind',
   752→    'lineage',
   753→    'scope',
   754→    'permissions',
   755→    'runtimeBindingHash',
   756→    'executionBindings',
   757→    'dependencyReceiptPolicy',
   758→    'candidateSemanticReviewPolicy',
   759→    'automationAuthorization',
   760→    'budget',
   761→    'acceptance',
   762→    'verificationPlan',
   763→    'failurePolicy',
   764→    ...(expectedVersion === 3 ? ['executionBoundary'] as const : []),
   765→  ] as const
   766→  assertExactKeys(record, allowedKeys, [
   767→    'version',
   768→    'objective',
   769→    'workflowKind',
   770→    'scope',
   771→    'permissions',
   772→    'executionBindings',
   773→    'budget',
   774→    'acceptance',
   775→    'verificationPlan',
   776→    'failurePolicy',
   777→    ...(expectedVersion === 3 ? ['executionBoundary'] as const : []),
   778→  ], 'contract')
   779→  if (record.version !== expectedVersion) {
   780→    failSchema('unsupported_version', 'contract.version', `仅支持版本 ${expectedVersion}`)
   781→  }
   782→  const executionBindings = readArray(record.executionBindings, 'contract.executionBindings')
   783→    .map((item, index) => readExecutionBindingV2(item, `contract.executionBindings[${index}]`))
   784→  if (executionBindings.length === 0) {
   785→    failSchema('invalid_execution_binding', 'contract.executionBindings', 'V2 正式契约必须包含 Skill 派生绑定')
   786→  }
   787→  assertUnique(executionBindings.map(item => item.stepId), 'contract.executionBindings')
   788→
   789→  const baseShape = { ...record, version: 1 } as Record<string, unknown>
   790→  delete baseShape.executionBindings
   791→  if (expectedVersion === 3) delete baseShape.executionBoundary
   792→  const base = parseAgentRunContractV1Base(baseShape, true)
   793→  const permissionSources = [...new Set(base.permissions.contextSourceKeys)].sort()
   794→  const bindingSources = [...new Set(executionBindings.flatMap(binding => binding.contextSourceKeys))].sort()
   795→  if (JSON.stringify(permissionSources) !== JSON.stringify(bindingSources)) {
   796→    failSchema(
   797→      'invalid_execution_binding',
   798→      'contract.permissions.contextSourceKeys',
   799→      'V2 Run permissions 必须等于所有 Skill binding 的实际来源并集',
   800→    )
   801→  }
   802→  const permissionWrites = writePermissionSignatures(base.permissions.writeTargets)
   803→  const bindingWrites = writePermissionSignatures(executionBindings.flatMap(binding => binding.writeTargets))
   804→  if (JSON.stringify(permissionWrites) !== JSON.stringify(bindingWrites)) {
   805→    failSchema(
   806→      'invalid_execution_binding',
   807→      'contract.permissions.writeTargets',
   808→      'V2 Run permissions 必须等于所有 Skill binding 的实际写目标并集',
   809→    )
   810→  }
   811→  if (executionBindings.some(binding => binding.maxOutputTokens > base.budget.maxOutputTokens)) {
   812→    failSchema(
   813→      'invalid_execution_binding',
   814→      'contract.budget.maxOutputTokens',
   815→      'Run 输出预算不得小于任一步骤的 Skill 输出预算',
   816→    )
   817→  }
   818→  return { base, executionBindings, record }
   819→}
