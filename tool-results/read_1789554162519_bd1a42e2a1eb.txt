  1942→function classifyWorkspaceMemory(spec: ProjectTableRegistration): WorkspaceMemoryClassificationV1 {
  1943→  if (spec.memoryClassification) return spec.memoryClassification
  1944→  if (spec.workspaceProjection) {
  1945→    return {
  1946→      version: 1,
  1947→      classification: 'editable',
  1948→      mirrorPolicy: 'editable-document',
  1949→      reason: '已登记严格 codec、稳定 identity、CAS 采纳与依赖影响发射器。',
  1950→    }
  1951→  }
  1952→  if (spec.owner === 'global') {
  1953→    return {
  1954→      version: 1,
  1955→      classification: 'not-applicable',
  1956→      mirrorPolicy: 'excluded-global',
  1957→      reason: '全局本地配置不属于任何项目工作区。',
  1958→    }
  1959→  }
  1960→  if (spec.exportable) {
  1961→    return {
  1962→      version: 1,
  1963→      classification: 'evidence',
  1964→      mirrorPolicy: 'recovery-evidence',
  1965→      reason: '确定内容进入注册表派生 recovery capsule；磁盘副本只读并可完整恢复。',
  1966→    }
  1967→  }
  1968→  return {
  1969→    version: 1,
  1970→    classification: 'derived-none',
  1971→    mirrorPolicy: 'rebuild-or-local-only',
  1972→    reason: '缓存、统计、临时事务或本地句柄不构成可移植项目记忆；由正式来源重建或留在本机。',
  1973→  }
  1974→}
