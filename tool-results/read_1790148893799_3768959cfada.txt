    66→    66→/* -------------------------------------------------------------------------- */
    67→    67→
    68→    68→function WeightBadge({ weight, total }: { weight: number; total: number }) {
    69→    69→  let color = 'var(--fg-muted)'
    70→    70→  let bg = 'transparent'
    71→    71→  if (weight >= 4) {
    72→    72→    color = 'var(--bg)'
    73→    73→    bg = 'var(--accent)'
    74→    74→  } else if (weight >= 2) {
    75→    75→    color = 'var(--accent)'
    76→    76→    bg = 'transparent'
    77→    77→  } else if (weight === 1) {
    78→    78→    color = 'var(--fg-faint)'
    79→    79→    bg = 'transparent'
    80→    80→  }
    81→    81→  return (
    82→    82→    <span
    83→    83→      className="font-mono-editorial text-[10px] px-1.5 py-0.5 rounded-sm border"
    84→    84→      style={{
    85→    85→        color,
    86→    86→        backgroundColor: bg,
    87→    87→        borderColor: `color-mix(in oklab, ${color} 35%, transparent)`,
    88→    88→      }}
    89→    89→      title={`Confirmed in ${weight} of ${total} distilled instances · 于 ${total} 个实例中确认 ${weight} 次`}
    90→    90→    >
    91→    91→      {weight}/{total}
    92→    92→    </span>
    93→    93→  )
    94→    94→}
