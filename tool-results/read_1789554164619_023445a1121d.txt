   171→async function requireTargetAdaptation(input: AssembleContextInput): Promise<AdaptationProject> {
   172→  if (input.adaptationProjectId == null || !input.scope) throw new Error('[adaptation-context] 缺少目标改编 selector')
   173→  const root = await db.adaptationProjects.get(input.adaptationProjectId)
   174→  if (!root
   175→    || root.projectId !== input.scope.projectId
   176→    || root.worldId !== input.scope.worldId
   177→    || root.workId !== input.scope.workId) {
   178→    throw new Error('[adaptation-context] 改编项目不属于当前目标 Work')
   179→  }
   180→  return root
   181→}

   183→async function readAdaptationSourceManifestContext(input: AssembleContextInput): Promise<string> {
   184→  const root = await requireTargetAdaptation(input)
   185→  const { inspectAdaptationFreshness, listActiveSourceUnits } = await loadAdaptationSourceReader()
   186→  const [units, freshness] = await Promise.all([
   187→    listActiveSourceUnits(root.id!),
   188→    inspectAdaptationFreshness(root.id!),
   189→  ])
   190→  return [
   191→    '【改编来源清单】',
   192→    `媒介：${root.medium}`,
   193→    `清单版本：v${root.activeSourceManifestVersion}`,
   194→    `清单哈希：${root.activeSourceManifestHash}`,
   195→    `来源覆盖：${root.sourceCoverage}`,
   196→    `来源状态：${freshness.status}`,
   197→    ...units.map(unit => `- ${unit.sourceUnitKey}｜${unit.sourceKind}｜${unit.label}｜顺序 ${unit.order}｜${unit.wordCount} 字｜hash ${unit.contentHash}｜摘要：${unit.summary}`),
   198→    ...(freshness.changes.length ? ['变化：', ...freshness.changes.map(change => `- ${change.kind}｜${change.sourceUnitKey}｜${change.label}`)] : []),
   199→  ].join('\n')
   200→}

   232→async function readAdaptationSourceFactsContext(input: AssembleContextInput): Promise<string> {
   233→  const root = await requireTargetAdaptation(input)
   234→  const { listAdaptationAnalysisV1 } = await import('../adaptation/analysis')
   235→  const analysis = await listAdaptationAnalysisV1({
   236→    scope: input.scope!,
   237→    adaptationProjectId: root.id!,
   238→    manifestVersion: root.activeSourceManifestVersion,
   239→  })
   240→  const facts = analysis.facts.filter(item => item.authorStatus === 'confirmed')
   241→  const edges = analysis.edges.filter(item => item.authorStatus === 'confirmed')
   242→  if (!facts.length && !edges.length) return ''
   243→  return [
   244→    `【已确认来源事实与因果图｜manifest v${root.activeSourceManifestVersion}】`,
   245→    ...facts.map(fact => `FACT ${fact.stableKey}｜${fact.kind}｜${fact.statement}｜subjects ${fact.subjectKeys.join(', ') || '-'}｜sources ${fact.sourceUnitKeys.join(', ')}`),
   246→    ...edges.map(edge => `EDGE ${edge.stableKey}｜${edge.fromFactKey} -${edge.relation}-> ${edge.toFactKey}｜${edge.rationale}｜sources ${edge.sourceUnitKeys.join(', ')}`),
   247→  ].join('\n')

  1054→/**
  1055→ * 智能截断函数：保留首尾，避免丢失反转信息
  1056→ * @param text 原始文本
  1057→ * @param maxChars 最大字符数
  1058→ * @returns 截断后的文本
  1059→ */
  1060→function smartTruncate(text: string, maxChars: number): string {
  1061→  if (!text || text.length <= maxChars) return text || ''
  1062→
  1063→  // 保留首段（通常是铺垫）
  1064→  const headSize = Math.floor(maxChars * 0.6)
  1065→  // 保留尾段（通常是反转/结果）
  1066→  const tailSize = Math.floor(maxChars * 0.4) - 5
  1067→
  1068→  const head = text.slice(0, headSize)
  1069→  const tail = text.slice(-tailSize)
  1070→
  1071→  return `${head}...${tail}`
  1072→}

  1437→/**
  1438→ * NS-4 · 当前有效事实投影（事实账本 → 生成上下文）。
  1439→ * 注入当前有效的 confirmed，以及在目标时点仍有效的 superseded 历史 Canon。
  1440→ * 按【规范章序】实时解析 validFrom/To（绝不缓存 order）判定"截止本章是否有效"，
  1441→ * 并按当前世界（∪ 默认 null 世界）过滤。这是事实账本改善长期一致性的回报通道。
  1442→ */
  1443→async function readCurrentFacts(projectId: number, chapterId?: number | null, worldGroupId?: number | null, scope?: WorkspaceScope): Promise<string> {
  1444→  if (chapterId == null) return ''
  1445→  const resolved = scope ?? await resolveScope({ projectId })
  1446→  const [facts, outlineNodes, chapters] = await Promise.all([
  1447→    readOwnedRows<any>(resolved, 'temporalFacts', { owner: 'work' }).then(rows => rows.filter(f => f.status === 'confirmed' || f.status === 'superseded')),
  1448→    readOwnedRows<OutlineNode>(resolved, 'outlineNodes', { owner: 'work' }),
  1449→    readOwnedRows<Chapter>(resolved, 'chapters', { owner: 'work' }),
  1450→  ])
  1451→  if (!facts.length) return ''
  1452→  const { sequence } = resolveCanonicalChapterSequence(outlineNodes, chapters)
  1453→  const orderOf = new Map<number, number>()
  1454→  sequence.forEach((entry, i) => { if (entry.chapter.id != null) orderOf.set(entry.chapter.id, i) })
  1455→  const currentOrder = orderOf.get(chapterId)
  1456→  if (currentOrder == null) return ''
  1457→
  1458→  const validNow = facts.filter(fact => {
  1459→    if (fact.worldGroupId != null && fact.worldGroupId !== (worldGroupId ?? null)) return false // 世界隔离
  1460→    const from = fact.validFromChapterId != null ? orderOf.get(fact.validFromChapterId) : -1
  1461→    if (from == null || from > currentOrder) return false   // 引用不存在的章 / 尚未生效
  1462→    if (fact.validToChapterId != null) {
  1463→      const to = orderOf.get(fact.validToChapterId)
  1464→      if (to != null && to <= currentOrder) return false     // 已失效
  1465→    }
  1466→    return true
  1467→  })
  1468→  if (!validNow.length) return ''
  1469→  const lines = validNow.slice(0, 80).map(fact => {
  1470→    const spec = getFactPredicate(fact.predicate)
  1471→    return `- ${fact.subjectName}｜${spec?.label ?? fact.predicate}：${fact.value}`
  1472→  })
  1473→  return ['【当前有效事实（截止本章·已确认，请勿与之矛盾）】', ...lines].join('\n')
  1474→}

  1481→async function readRetrievedPassages(projectId: number, chapterId?: number | null, outlineNodeId?: number | null, worldGroupId?: number | null, scope?: WorkspaceScope): Promise<string> {

  1497→  const embCfg = useAIConfigStore.getState().embedding

  1511→  return [hierarchy, '【相关前文召回（防止远距离细节/伏笔矛盾，仅供参考）】', ...lines].filter(Boolean).join('\n\n')
  1512→}
